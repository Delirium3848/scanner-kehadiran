async function onScanSuccess(qrText) { if (sedangProses) return; sedangProses = true; try { const url = API + "?id=" + encodeURIComponent(qrText) + "&t=" + Date.now(); const res = await fetch(url); const data = await res.json(); if (data.status === "BERJAYA") { // Papar nama (jika ada) serta masa imbasan dalam popup & suara const infoNama = data.nama ? `
${data.nama}` : ''; const infoMasa = `
Masa: ${data.masa}`; cakap("Thank you. Your attendance has been recorded."); popup(`✅
RECORDED SUCCESSFUL${infoNama}${infoMasa}`, "success"); } else if (data.status === "SUDAH_HADIR") { const infoNama = data.nama ? `
${data.nama}` : ''; cakap("You have already been recorded as present today."); popup(`⚠️
ALREADY PRESENT${infoNama}`, "warning"); } else if (data.status === "TIDAK_DIJUMPAI") { cakap("Sorry. This QR code is not recognized."); popup("❌
QR CODE NOT RECOGNIZED", "danger"); } else { cakap("Sorry. There was a problem recording your attendance."); popup("❌
RECORDING ERROR", "danger"); } } catch (err) { console.log("SCAN ERROR:", err); cakap("Sorry. There was a problem connecting to the attendance system."); popup("❌
CONNECTION ERROR", "danger"); } setTimeout(() => { sedangProses = false; }, 2000); }
